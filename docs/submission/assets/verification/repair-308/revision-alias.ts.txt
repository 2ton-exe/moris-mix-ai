// Synthetic regression diagnostic; no provider or database access.
import { assertEquals } from 'jsr:@std/assert@1';
import { explainRecipeRevision, type RecipeRevisionContext } from './supabase/functions/_shared/ai/recipeRevision.ts';
import { GeneratedRevisionSchema } from './supabase/functions/_shared/ai/schemas.ts';
const canonical = {
  id: '11111111-1111-4111-8111-111111111111', title: 'Synthetic tomato rice', star_rating: null,
  ingredients: [
    { name: 'rice', quantity: 1, unit: 'cup', inferred: false },
    { name: 'tomato', quantity: 1, unit: 'piece', inferred: false },
    { name: 'olive oil', quantity: 1, unit: 'tsp', inferred: false },
  ],
  instructions: [{ step: 1, text: 'Warm olive oil and tomato.' }, { step: 2, text: 'Stir in cooked rice.' }],
};
const noteExcerpt = 'Change olive oil from 1 teaspoon to 2 teaspoons while keeping rice and tomato unchanged.';
const output = GeneratedRevisionSchema.parse({
  recipe: { ...canonical, ingredients: canonical.ingredients.map((item, index) => index === 2 ? { ...item, quantity: 2, unit: 'teaspoons' } : item) },
  substitutions: [], parentRecipeIds: [canonical.id], revisionReason: 'Use two teaspoons of olive oil as requested.',
});
Deno.test('control: canonical unchanged rice cup permits oil 1 tsp to 2 tsp', () => {
  assertEquals(explainRecipeRevision({ source: canonical, noteExcerpt }, output)?.after, { quantity: 2, unit: 'tsp' });
});
Deno.test('legacy equivalent rice cups to cup must not block oil 1 tsp to 2 tsp', () => {
  const source = structuredClone(canonical);
  source.ingredients[0].unit = 'cups';
  const context = { source, noteExcerpt } satisfies RecipeRevisionContext;
  assertEquals(explainRecipeRevision(context, output)?.after, { quantity: 2, unit: 'tsp' });
});
Deno.test('control: equivalent oil amount is not a real change', () => {
  const equivalent = structuredClone(output);
  equivalent.recipe.ingredients[2] = { name: 'olive oil', quantity: 1 / 3, unit: 'tbsp', inferred: false };
  assertEquals(explainRecipeRevision({ source: canonical, noteExcerpt }, equivalent), null);
});
